CREATE TABLE IF NOT EXISTS pelanggan (
    id_pelanggan BIGINT
        GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nama_pelanggan VARCHAR(100) NOT NULL,
    no_hp          VARCHAR(20),
    alamat         TEXT,
    created_at     TIMESTAMPTZ DEFAULT NOW()
);

ALTER TABLE public.pelanggan
    DROP COLUMN IF EXISTS kategori;


-- =====================================================
-- 2. TABEL PERSEDIAAN
-- =====================================================

CREATE TABLE IF NOT EXISTS persediaan (
    id_persediaan BIGINT
        GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    nama_barang VARCHAR(100) NOT NULL,
    stok        INTEGER NOT NULL DEFAULT 0,
    harga       NUMERIC(12,2) NOT NULL DEFAULT 0,
    satuan      VARCHAR(30) NOT NULL DEFAULT 'Galon',
    updated_at  TIMESTAMPTZ DEFAULT NOW(),
    CONSTRAINT cek_stok CHECK (stok >= 0),
    CONSTRAINT cek_harga CHECK (harga >= 0)
);


-- =====================================================
-- 3. TABEL PENJUALAN
-- =====================================================

CREATE TABLE IF NOT EXISTS penjualan (
    id_penjualan      BIGINT
        GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    id_pelanggan      BIGINT NOT NULL,
    id_persediaan     BIGINT NOT NULL,
    tanggal_penjualan DATE NOT NULL DEFAULT CURRENT_DATE,
    jumlah_galon      INTEGER NOT NULL,
    harga_per_galon   NUMERIC(12,2) NOT NULL,
    total_penjualan   NUMERIC(12,2) NOT NULL DEFAULT 0,
    status_pembayaran VARCHAR(20) NOT NULL DEFAULT 'Belum Lunas',
    jenis_pembayaran  VARCHAR(10) NOT NULL DEFAULT 'Kredit',
    status_penyerahan VARCHAR(20) NOT NULL DEFAULT 'Belum Diserahkan',
    created_at        TIMESTAMPTZ DEFAULT NOW(),

    CONSTRAINT fk_penjualan_pelanggan
        FOREIGN KEY (id_pelanggan)
        REFERENCES pelanggan(id_pelanggan)
            ON UPDATE CASCADE ON DELETE RESTRICT,

    CONSTRAINT fk_penjualan_persediaan
        FOREIGN KEY (id_persediaan)
        REFERENCES persediaan(id_persediaan)
            ON UPDATE CASCADE ON DELETE RESTRICT,

    CONSTRAINT cek_jumlah_galon    CHECK (jumlah_galon > 0),
    CONSTRAINT cek_harga_galon     CHECK (harga_per_galon >= 0),
    CONSTRAINT cek_status_pembayaran
        CHECK (status_pembayaran IN ('Lunas', 'Belum Lunas')),
    CONSTRAINT cek_jenis_pembayaran
        CHECK (jenis_pembayaran IN ('Tunai', 'Kredit')),
    CONSTRAINT cek_status_penyerahan
        CHECK (status_penyerahan IN ('Belum Diserahkan', 'Sudah Sampai'))
);

ALTER TABLE public.penjualan
    ADD COLUMN IF NOT EXISTS jenis_pembayaran VARCHAR(10) NOT NULL DEFAULT 'Kredit',
    ADD COLUMN IF NOT EXISTS status_penyerahan VARCHAR(20) NOT NULL DEFAULT 'Belum Diserahkan';

ALTER TABLE public.penjualan
    DROP CONSTRAINT IF EXISTS cek_jenis_pembayaran,
    ADD CONSTRAINT cek_jenis_pembayaran
        CHECK (jenis_pembayaran IN ('Tunai', 'Kredit')),
    DROP CONSTRAINT IF EXISTS cek_status_penyerahan,
    ADD CONSTRAINT cek_status_penyerahan
        CHECK (status_penyerahan IN ('Belum Diserahkan', 'Sudah Sampai'));

CREATE INDEX IF NOT EXISTS idx_penjualan_pelanggan   ON penjualan(id_pelanggan);
CREATE INDEX IF NOT EXISTS idx_penjualan_persediaan  ON penjualan(id_persediaan);
CREATE INDEX IF NOT EXISTS idx_penjualan_tanggal     ON penjualan(tanggal_penjualan);

DO $$
DECLARE
    constraint_row RECORD;
BEGIN
    FOR constraint_row IN
        SELECT c.conname
        FROM pg_constraint AS c
        JOIN pg_attribute AS a
          ON a.attrelid = c.conrelid
         AND a.attnum = ANY(c.conkey)
        WHERE c.contype = 'c'
          AND c.conrelid = 'public.penerimaan_kas'::regclass
          AND a.attname = 'metode_pembayaran'
    LOOP
        EXECUTE format(
            'ALTER TABLE public.penerimaan_kas DROP CONSTRAINT %I',
            constraint_row.conname
        );
    END LOOP;
END $$;

-- 5a. BEFORE INSERT: validasi stok + hitung total
CREATE OR REPLACE FUNCTION fn_penjualan_before_insert()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
DECLARE
    stok_sekarang INTEGER;
BEGIN
    IF NEW.jenis_pembayaran = 'Tunai' THEN
        NEW.status_pembayaran := 'Lunas';
    ELSE
        NEW.status_pembayaran := 'Belum Lunas';
    END IF;

    SELECT stok INTO stok_sekarang
    FROM persediaan
    WHERE id_persediaan = NEW.id_persediaan;

    IF stok_sekarang IS NULL THEN
        RAISE EXCEPTION 'Persediaan dengan id % tidak ditemukan',
            NEW.id_persediaan;
    END IF;

    IF stok_sekarang < NEW.jumlah_galon THEN
        RAISE EXCEPTION
            'Stok tidak mencukupi. Tersedia %, diminta %',
            stok_sekarang, NEW.jumlah_galon;
    END IF;

    NEW.total_penjualan := NEW.jumlah_galon * NEW.harga_per_galon;

    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_penjualan_before_insert ON penjualan;
CREATE TRIGGER trg_penjualan_before_insert
BEFORE INSERT ON penjualan
FOR EACH ROW
EXECUTE FUNCTION fn_penjualan_before_insert();


-- 5b. AFTER INSERT: kurangi stok dan catat pembayaran tunai
CREATE OR REPLACE FUNCTION fn_penjualan_after_insert()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE persediaan
    SET stok = stok - NEW.jumlah_galon,
        updated_at = NOW()
    WHERE id_persediaan = NEW.id_persediaan;

    IF NEW.jenis_pembayaran = 'Tunai' AND NEW.total_penjualan > 0 THEN
        INSERT INTO penerimaan_kas (
            id_penjualan,
            tanggal_penerimaan,
            jumlah_diterima,
            metode_pembayaran
        ) VALUES (
            NEW.id_penjualan,
            NEW.tanggal_penjualan,
            NEW.total_penjualan,
            'Tunai'
        );
    END IF;

    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_penjualan_after_insert ON penjualan;
CREATE TRIGGER trg_penjualan_after_insert
AFTER INSERT ON penjualan
FOR EACH ROW
EXECUTE FUNCTION fn_penjualan_after_insert();


-- 5c. AFTER DELETE: kembalikan stok
CREATE OR REPLACE FUNCTION fn_penjualan_after_delete()
RETURNS TRIGGER
LANGUAGE plpgsql
AS $$
BEGIN
    UPDATE persediaan
    SET stok = stok + OLD.jumlah_galon,
        updated_at = NOW()
    WHERE id_persediaan = OLD.id_persediaan;

    RETURN OLD;
END;
$$;

DROP TRIGGER IF EXISTS trg_penjualan_after_delete ON penjualan;
CREATE TRIGGER trg_penjualan_after_delete
AFTER DELETE ON penjualan
FOR EACH ROW
EXECUTE FUNCTION fn_penjualan_after_delete();

-- Sinkronkan status pelunasan dari seluruh penerimaan kas.
CREATE OR REPLACE FUNCTION public.fn_sinkron_status_pembayaran()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
    target_id_penjualan BIGINT;
BEGIN
    IF TG_OP <> 'INSERT'
       AND TG_OP <> 'DELETE'
       AND OLD.id_penjualan IS DISTINCT FROM NEW.id_penjualan THEN
        UPDATE public.penjualan AS p
        SET status_pembayaran = CASE
            WHEN p.total_penjualan <= COALESCE((
                SELECT SUM(pk.jumlah_diterima)
                FROM public.penerimaan_kas AS pk
                WHERE pk.id_penjualan = p.id_penjualan
            ), 0) THEN 'Lunas'
            ELSE 'Belum Lunas'
        END
        WHERE p.id_penjualan = OLD.id_penjualan;
    END IF;

    IF TG_OP = 'DELETE' THEN
        target_id_penjualan := OLD.id_penjualan;
    ELSE
        target_id_penjualan := NEW.id_penjualan;
    END IF;

    UPDATE public.penjualan AS p
    SET status_pembayaran = CASE
        WHEN p.total_penjualan <= COALESCE((
            SELECT SUM(pk.jumlah_diterima)
            FROM public.penerimaan_kas AS pk
            WHERE pk.id_penjualan = p.id_penjualan
        ), 0) THEN 'Lunas'
        ELSE 'Belum Lunas'
    END
    WHERE p.id_penjualan = target_id_penjualan;

    IF TG_OP = 'DELETE' THEN
        RETURN OLD;
    END IF;
    RETURN NEW;
END;
$$;

DROP TRIGGER IF EXISTS trg_penerimaan_sinkron_status ON public.penerimaan_kas;
CREATE TRIGGER trg_penerimaan_sinkron_status
AFTER INSERT OR UPDATE OR DELETE ON public.penerimaan_kas
FOR EACH ROW
EXECUTE FUNCTION public.fn_sinkron_status_pembayaran();

-- Catat pembayaran dan perbarui status dalam satu transaksi atomik.
CREATE OR REPLACE FUNCTION public.fn_catat_penerimaan(
    p_id_penjualan BIGINT,
    p_tanggal_penerimaan DATE,
    p_jumlah_diterima NUMERIC,
    p_metode_pembayaran TEXT
)
RETURNS BIGINT
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
    total_tagihan NUMERIC(12,2);
    total_dibayar NUMERIC(12,2);
    id_penerimaan_baru BIGINT;
BEGIN
    IF p_jumlah_diterima IS NULL OR p_jumlah_diterima <= 0 THEN
        RAISE EXCEPTION 'Jumlah diterima harus lebih besar dari nol';
    END IF;

    IF p_metode_pembayaran IS NULL
       OR p_metode_pembayaran NOT IN ('Tunai', 'Transfer', 'QRIS') THEN
        RAISE EXCEPTION 'Metode pembayaran tidak valid';
    END IF;

    SELECT p.total_penjualan
    INTO total_tagihan
    FROM public.penjualan AS p
    WHERE p.id_penjualan = p_id_penjualan
    FOR UPDATE;

    IF NOT FOUND THEN
        RAISE EXCEPTION 'Transaksi penjualan tidak ditemukan';
    END IF;

    SELECT COALESCE(SUM(pk.jumlah_diterima), 0)
    INTO total_dibayar
    FROM public.penerimaan_kas AS pk
    WHERE pk.id_penjualan = p_id_penjualan;

    IF p_jumlah_diterima > total_tagihan - total_dibayar THEN
        RAISE EXCEPTION 'Jumlah diterima melebihi sisa piutang';
    END IF;

    INSERT INTO public.penerimaan_kas (
        id_penjualan,
        tanggal_penerimaan,
        jumlah_diterima,
        metode_pembayaran
    ) VALUES (
        p_id_penjualan,
        COALESCE(p_tanggal_penerimaan, CURRENT_DATE),
        p_jumlah_diterima,
        p_metode_pembayaran
    )
    RETURNING id_penerimaan INTO id_penerimaan_baru;

    RETURN id_penerimaan_baru;
END;
$$;

-- =====================================================
-- 6. VIEW MENU PIUTANG DAN PENYERAHAN BARANG
-- =====================================================

CREATE OR REPLACE VIEW public.v_piutang
WITH (security_invoker = true)
AS
WITH pembayaran AS (
    SELECT id_penjualan, SUM(jumlah_diterima) AS total_dibayar
    FROM public.penerimaan_kas
    GROUP BY id_penjualan
)
SELECT
    p.id_penjualan,
    p.tanggal_penjualan,
    p.id_pelanggan,
    pl.nama_pelanggan,
    p.total_penjualan,
    COALESCE(byr.total_dibayar, 0)::NUMERIC(12,2) AS total_dibayar,
    GREATEST(
        p.total_penjualan - COALESCE(byr.total_dibayar, 0),
        0
    )::NUMERIC(12,2) AS sisa_piutang
FROM public.penjualan AS p
JOIN public.pelanggan AS pl ON pl.id_pelanggan = p.id_pelanggan
LEFT JOIN pembayaran AS byr ON byr.id_penjualan = p.id_penjualan
WHERE p.total_penjualan > COALESCE(byr.total_dibayar, 0);

CREATE OR REPLACE VIEW public.v_penyerahan_barang
WITH (security_invoker = true)
AS
WITH pembayaran AS (
    SELECT id_penjualan, SUM(jumlah_diterima) AS total_dibayar
    FROM public.penerimaan_kas
    GROUP BY id_penjualan
)
SELECT
    p.id_penjualan,
    p.tanggal_penjualan,
    p.id_pelanggan,
    pl.nama_pelanggan,
    p.id_persediaan,
    ps.nama_barang,
    p.jumlah_galon,
    p.total_penjualan,
    COALESCE(byr.total_dibayar, 0)::NUMERIC(12,2) AS total_dibayar,
    GREATEST(
        p.total_penjualan - COALESCE(byr.total_dibayar, 0),
        0
    )::NUMERIC(12,2) AS sisa_piutang,
    CASE
        WHEN p.total_penjualan <= COALESCE(byr.total_dibayar, 0) THEN 'Lunas'
        ELSE 'Belum Lunas'
    END AS status_pembayaran,
    p.jenis_pembayaran,
    p.status_penyerahan
FROM public.penjualan AS p
JOIN public.pelanggan AS pl ON pl.id_pelanggan = p.id_pelanggan
JOIN public.persediaan AS ps ON ps.id_persediaan = p.id_persediaan
LEFT JOIN pembayaran AS byr ON byr.id_penjualan = p.id_penjualan;

-- =====================================================

-- Sebelum aplikasi digunakan, buat pengguna di Supabase Dashboard:
-- Authentication > Users, menggunakan email yang sama dengan AUTH_EMAIL
-- pada Frontend/JS/Config.js. Password dikelola oleh Supabase Auth, bukan SQL.

ALTER TABLE penerimaan_kas  ENABLE ROW LEVEL SECURITY;

GRANT USAGE ON SCHEMA public TO authenticated;
GRANT SELECT, INSERT, UPDATE, DELETE ON TABLE
    public.pelanggan,
    public.persediaan,
    public.penjualan,
    public.penerimaan_kas
TO authenticated;
GRANT SELECT ON TABLE
    public.v_piutang,
    public.v_penyerahan_barang
TO authenticated;
REVOKE ALL ON FUNCTION public.fn_catat_penerimaan(BIGINT, DATE, NUMERIC, TEXT) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.fn_catat_penerimaan(BIGINT, DATE, NUMERIC, TEXT)
TO authenticated;
REVOKE ALL ON FUNCTION public.fn_sinkron_status_pembayaran() FROM PUBLIC;

DO $$
DECLARE
    table_name TEXT;
    sequence_name REGCLASS;
    column_name TEXT;
    maximum_id BIGINT;
    sequence_last_value BIGINT;
    sequence_is_called BOOLEAN;
BEGIN
    FOREACH table_name IN ARRAY ARRAY[
        'pelanggan', 'persediaan', 'penjualan', 'penerimaan_kas'
    ]
    LOOP
        column_name := CASE table_name
            WHEN 'pelanggan' THEN 'id_pelanggan'
            WHEN 'persediaan' THEN 'id_persediaan'
            WHEN 'penjualan' THEN 'id_penjualan'
            ELSE 'id_penerimaan'
        END;
        sequence_name := pg_get_serial_sequence(
            format('public.%I', table_name),
            column_name
        )::REGCLASS;
        IF sequence_name IS NOT NULL THEN
            EXECUTE format(
                'REVOKE ALL ON SEQUENCE %s FROM PUBLIC, anon',
                sequence_name
            );
            EXECUTE format(
                'GRANT USAGE, SELECT ON SEQUENCE %s TO authenticated',
                sequence_name
            );
            EXECUTE format(
                'SELECT MAX(%I) FROM public.%I',
                column_name,
                table_name
            ) INTO maximum_id;
            EXECUTE format(
                'SELECT last_value, is_called FROM %s',
                sequence_name
            ) INTO sequence_last_value, sequence_is_called;
            IF maximum_id IS NOT NULL OR sequence_is_called THEN
                PERFORM setval(
                    sequence_name,
                    GREATEST(
                        COALESCE(maximum_id, 1),
                        CASE WHEN sequence_is_called THEN sequence_last_value ELSE 1 END
                    ),
                    TRUE
                );
            END IF;
        END IF;
    END LOOP;

    IF to_regclass('public.pengeluaran_kas') IS NOT NULL THEN
        EXECUTE 'ALTER TABLE public.pengeluaran_kas ENABLE ROW LEVEL SECURITY';
        EXECUTE 'DROP POLICY IF EXISTS public_all_pengeluaran ON public.pengeluaran_kas';
        EXECUTE 'REVOKE ALL ON TABLE public.pengeluaran_kas FROM PUBLIC, anon, authenticated';

        sequence_name := pg_get_serial_sequence(
            'public.pengeluaran_kas',
            'id_pengeluaran'
        )::REGCLASS;
        IF sequence_name IS NOT NULL THEN
            EXECUTE format(
                'REVOKE ALL ON SEQUENCE %s FROM PUBLIC, anon, authenticated',
                sequence_name
            );
        END IF;
    END IF;
END $$;

-- Hapus policy lama agar tidak ada policy permissive yang tetap membuka akses.
DO $$
DECLARE
    target_table TEXT;
    policy_name TEXT;
BEGIN
    FOREACH target_table IN ARRAY ARRAY[
        'pelanggan', 'persediaan', 'penjualan', 'penerimaan_kas'
    ]
    LOOP
        FOR policy_name IN
            SELECT polname
            FROM pg_policy
            WHERE polrelid = format('public.%I', target_table)::REGCLASS
        LOOP
            EXECUTE format(
                'DROP POLICY %I ON public.%I',
                policy_name,
                target_table
            );
        END LOOP;
    END LOOP;
END $$;

CREATE OR REPLACE FUNCTION public.fn_hitung_hpp_produk(p_produk_id BIGINT)
RETURNS VOID
LANGUAGE SQL
SECURITY INVOKER
SET search_path = ''
AS $$
    UPDATE public.produk AS p
    SET hpp_per_unit = COALESCE((
        SELECT ROUND(SUM(k.jumlah_pakai * b.harga_modal_satuan), 2)
        FROM public.komposisi_produk AS k
        JOIN public.bahan_persediaan AS b ON b.id = k.bahan_id
        WHERE k.produk_id = p.id
    ), 0),
        updated_at = NOW()
    WHERE p.id = p_produk_id;
$$;

CREATE OR REPLACE FUNCTION public.fn_komposisi_refresh_hpp()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
BEGIN
    IF TG_OP <> 'INSERT' THEN
        PERFORM public.fn_hitung_hpp_produk(OLD.produk_id);
    END IF;
    IF TG_OP <> 'DELETE'
       AND (TG_OP = 'INSERT' OR NEW.produk_id IS DISTINCT FROM OLD.produk_id) THEN
        PERFORM public.fn_hitung_hpp_produk(NEW.produk_id);
    END IF;
    IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
    RETURN NEW;
END;
$$;
DROP TRIGGER IF EXISTS trg_komposisi_refresh_hpp ON public.komposisi_produk;
CREATE TRIGGER trg_komposisi_refresh_hpp
AFTER INSERT OR UPDATE OR DELETE ON public.komposisi_produk
FOR EACH ROW EXECUTE FUNCTION public.fn_komposisi_refresh_hpp();

CREATE OR REPLACE FUNCTION public.fn_bahan_refresh_hpp()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE target_produk_id BIGINT;
BEGIN
    FOR target_produk_id IN
        SELECT k.produk_id FROM public.komposisi_produk AS k
        WHERE k.bahan_id = NEW.id
    LOOP
        PERFORM public.fn_hitung_hpp_produk(target_produk_id);
    END LOOP;
    RETURN NEW;
END;
$$;
DROP TRIGGER IF EXISTS trg_bahan_refresh_hpp ON public.bahan_persediaan;
CREATE TRIGGER trg_bahan_refresh_hpp
AFTER UPDATE OF harga_modal_satuan ON public.bahan_persediaan
FOR EACH ROW
WHEN (OLD.harga_modal_satuan IS DISTINCT FROM NEW.harga_modal_satuan)
EXECUTE FUNCTION public.fn_bahan_refresh_hpp();

UPDATE public.produk AS p

    JOIN public.bahan_persediaan AS b ON b.id = k.bahan_id
    WHERE k.produk_id = p.id

    updated_at = NOW();

CREATE OR REPLACE FUNCTION public.fn_penjualan_before_insert()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
    product_name TEXT;
    recipe RECORD;
    hpp_unit NUMERIC(14,4) := 0;
    jumlah_resep INTEGER := 0;
    jumlah_dibutuhkan NUMERIC(14,4);
BEGIN
    NEW.status_pembayaran := CASE
        WHEN NEW.jenis_pembayaran = 'Tunai' THEN 'Lunas'
        ELSE 'Belum Lunas'
    END;

    SELECT p.nama_produk INTO product_name
    FROM public.produk AS p
    WHERE p.id = NEW.produk_id
    FOR UPDATE;
    IF NOT FOUND THEN
        RAISE EXCEPTION 'Produk dengan id % tidak ditemukan', NEW.produk_id;
    END IF;

    FOR recipe IN
        SELECT k.bahan_id, k.jumlah_pakai, b.stok_tersedia, b.harga_modal_satuan
        FROM public.komposisi_produk AS k
        JOIN public.bahan_persediaan AS b ON b.id = k.bahan_id
        WHERE k.produk_id = NEW.produk_id
        ORDER BY k.bahan_id
        FOR UPDATE OF k, b
    LOOP
        jumlah_resep := jumlah_resep + 1;
        jumlah_dibutuhkan := recipe.jumlah_pakai * NEW.jumlah_galon;
        IF recipe.stok_tersedia < jumlah_dibutuhkan THEN
            RAISE EXCEPTION 'Stok bahan % tidak mencukupi. Tersedia %, dibutuhkan %',
                recipe.bahan_id, recipe.stok_tersedia, jumlah_dibutuhkan;
        END IF;
        hpp_unit := hpp_unit
            + recipe.jumlah_pakai * recipe.harga_modal_satuan;
    END LOOP;
    IF jumlah_resep = 0 THEN
        RAISE EXCEPTION 'Komposisi produk id % belum diatur',
            NEW.produk_id;
    END IF;

    NEW.total_penjualan := NEW.jumlah_galon * NEW.harga_per_galon;
    hpp_unit := ROUND(hpp_unit, 2);
    NEW.total_hpp := hpp_unit * NEW.jumlah_galon;
    NEW.laba := NEW.total_penjualan - NEW.total_hpp;
    RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.fn_penjualan_after_insert()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE new_detail_id BIGINT;
BEGIN
    INSERT INTO public.detail_penjualan (
        id_penjualan, produk_id, jumlah, hpp_satuan
    ) VALUES (
        NEW.id_penjualan, NEW.produk_id, NEW.jumlah_galon,
        ROUND(NEW.total_hpp / NEW.jumlah_galon, 2)
    )
    RETURNING id_detail INTO new_detail_id;

    INSERT INTO public.detail_penjualan_bahan (
        detail_id, bahan_id, jumlah_pakai_saat_transaksi
    )
    SELECT new_detail_id, k.bahan_id, k.jumlah_pakai
    FROM public.komposisi_produk AS k
    WHERE k.produk_id = NEW.produk_id;

    UPDATE public.bahan_persediaan AS b
    SET stok_tersedia = b.stok_tersedia - pemakaian.jumlah_dipakai,
        updated_at = NOW()
    FROM (
        SELECT k.bahan_id, SUM(k.jumlah_pakai * NEW.jumlah_galon) AS jumlah_dipakai
        FROM public.komposisi_produk AS k
        WHERE k.produk_id = NEW.produk_id
        GROUP BY k.bahan_id
    ) AS pemakaian
    WHERE b.id = pemakaian.bahan_id;

    IF NEW.jenis_pembayaran = 'Tunai' AND NEW.total_penjualan > 0 THEN
        INSERT INTO public.penerimaan_kas (
            id_penjualan, tanggal_penerimaan, jumlah_diterima, metode_pembayaran
        ) VALUES (
            NEW.id_penjualan, NEW.tanggal_penjualan, NEW.total_penjualan, 'Tunai'
        );
    END IF;
    RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.fn_penjualan_restore_bahan()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
BEGIN
    UPDATE public.bahan_persediaan AS b
    SET stok_tersedia = b.stok_tersedia
            + pemakaian.jumlah_pakai_saat_transaksi * d.jumlah,
        updated_at = NOW()
    FROM public.detail_penjualan AS d
    JOIN public.detail_penjualan_bahan AS pemakaian
      ON pemakaian.detail_id = d.id_detail
    WHERE d.id_penjualan = OLD.id_penjualan
      AND b.id = pemakaian.bahan_id;
    RETURN OLD;
END;
$$;
DROP TRIGGER IF EXISTS trg_penjualan_restore_bahan ON public.penjualan;
CREATE TRIGGER trg_penjualan_restore_bahan
BEFORE DELETE ON public.penjualan
FOR EACH ROW EXECUTE FUNCTION public.fn_penjualan_restore_bahan();

CREATE OR REPLACE FUNCTION public.fn_penjualan_after_delete()
RETURNS TRIGGER
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
BEGIN
    IF OLD.id_persediaan IS NOT NULL THEN
        UPDATE public.persediaan
        SET stok = stok + OLD.jumlah_galon,
            updated_at = NOW()
        WHERE id_persediaan = OLD.id_persediaan;
    END IF;
    RETURN OLD;
END;
$$;

CREATE OR REPLACE FUNCTION public.fn_catat_kulakan(
    p_bahan_id BIGINT, p_jumlah NUMERIC, p_harga_satuan NUMERIC
)
RETURNS BIGINT
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path = ''
AS $$
DECLARE
    bahan RECORD;
    stok_baru NUMERIC(14,4);
    harga_baru NUMERIC(12,2);
BEGIN
    IF p_jumlah IS NULL OR p_jumlah <= 0 THEN
        RAISE EXCEPTION 'Jumlah stok masuk harus lebih besar dari nol';
    END IF;
    IF p_harga_satuan IS NULL OR p_harga_satuan < 0 THEN
        RAISE EXCEPTION 'Harga modal satuan tidak valid';
    END IF;

    SELECT b.id, b.stok_tersedia, b.harga_modal_satuan INTO bahan
    FROM public.bahan_persediaan AS b
    WHERE b.id = p_bahan_id
    FOR UPDATE;
    IF NOT FOUND THEN RAISE EXCEPTION 'Bahan persediaan tidak ditemukan'; END IF;

    stok_baru := bahan.stok_tersedia + p_jumlah;
    harga_baru := ROUND(
        (bahan.stok_tersedia * bahan.harga_modal_satuan
            + p_jumlah * p_harga_satuan) / stok_baru,
        2
    );
    UPDATE public.bahan_persediaan
    SET stok_tersedia = stok_baru,
        harga_modal_satuan = harga_baru,
        updated_at = NOW()
    WHERE id = p_bahan_id;
    RETURN p_bahan_id;
END;
$$;

CREATE OR REPLACE VIEW public.v_penyerahan_barang
WITH (security_invoker = true)
AS
WITH pembayaran AS (
    SELECT id_penjualan, SUM(jumlah_diterima) AS total_dibayar
    FROM public.penerimaan_kas
    GROUP BY id_penjualan
)
SELECT
    p.id_penjualan,
    p.tanggal_penjualan,
    p.id_pelanggan,
    pl.nama_pelanggan,
    p.produk_id AS id_persediaan,
    pr.nama_produk::VARCHAR(100) AS nama_barang,
    p.jumlah_galon,
    p.total_penjualan,
    COALESCE(byr.total_dibayar, 0)::NUMERIC(12,2) AS total_dibayar,
    GREATEST(p.total_penjualan - COALESCE(byr.total_dibayar, 0), 0)::NUMERIC(12,2)
        AS sisa_piutang,
    CASE
        WHEN p.total_penjualan <= COALESCE(byr.total_dibayar, 0) THEN 'Lunas'
        ELSE 'Belum Lunas'
    END AS status_pembayaran,
    p.jenis_pembayaran,
    p.status_penyerahan
FROM public.penjualan AS p
JOIN public.pelanggan AS pl ON pl.id_pelanggan = p.id_pelanggan
JOIN public.produk AS pr ON pr.id = p.produk_id
LEFT JOIN pembayaran AS byr ON byr.id_penjualan = p.id_penjualan;

-- Harapkan rls_enabled dan authenticated_can_select bernilai true untuk
-- seluruh tabel HPP. Akses tulis dibatasi policy akun aplikasi terautentikasi.
SELECT c.relname AS table_name,
       c.relrowsecurity AS rls_enabled,
       has_table_privilege('authenticated', c.oid, 'SELECT')
           AS authenticated_can_select
FROM pg_class AS c
JOIN pg_namespace AS n ON n.oid = c.relnamespace
WHERE n.nspname = 'public'
  AND c.relname IN (
      'produk',
      'bahan_persediaan',
      'komposisi_produk',
      'detail_penjualan',
      'detail_penjualan_bahan'
  )
ORDER BY c.relname;
